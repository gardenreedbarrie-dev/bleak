file(GLOB PRIVATE_SOURCES CONFIGURE_DEPENDS "*.cpp")

# BLEAKHOLD(bleakfx): the shaders in Shaders/ are compiled with the Windows SDK's fxc and embedded as byte arrays
# (BleakFxShaders.h). Without fxc, BleakFx is built without shaders and stays off; the client still builds, unless
# BLEAKFX_REQUIRE_SHADERS is on (CI).
option(BLEAKFX_REQUIRE_SHADERS "Stop the configure when fxc is missing instead of building BleakFx without shaders" OFF)
set(BLEAKFX_SHADERS
    bh_fullscreen_vs:vs_3_0
    bh_bloom_pre_ps:ps_3_0
    bh_bloom_down_ps:ps_3_0
    bh_bloom_up_ps:ps_3_0
    bh_composite_ps:ps_3_0
    bh_fxaa_ps:ps_3_0
    bh_cas_ps:ps_3_0
    bh_ao_ps:ps_3_0
    bh_ao_blur_ps:ps_3_0
    bh_rays_mask_ps:ps_3_0
    bh_rays_blur_ps:ps_3_0
    bh_depthfx_ps:ps_3_0
    bh_depthcopy_ps:ps_3_0
    bh_water_vs:vs_3_0
    bh_water_ps:ps_3_0
    bh_clear_ps:ps_3_0
    bh_smaa_edge_ps:ps_3_0
    bh_smaa_weights_ps:ps_3_0
    bh_smaa_blend_ps:ps_3_0
)

# What the shaders include.
set(BLEAKFX_SHADER_INCLUDES
    ${CMAKE_CURRENT_SOURCE_DIR}/Shaders/bh_common.hlsli
    ${CMAKE_CURRENT_SOURCE_DIR}/Shaders/bh_smaa_common.hlsli
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../ThirdParty/SMAA/SMAA.hlsl
)

set(BLEAKFX_FXC_HINTS)
if (CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION)
    list(APPEND BLEAKFX_FXC_HINTS "C:/Program Files (x86)/Windows Kits/10/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64")
endif ()
if (DEFINED ENV{WindowsSdkVerBinPath})
    list(APPEND BLEAKFX_FXC_HINTS "$ENV{WindowsSdkVerBinPath}/x64")
endif ()
file(GLOB BLEAKFX_SDK_BIN_DIRS LIST_DIRECTORIES true "C:/Program Files (x86)/Windows Kits/10/bin/10.*")
list(SORT BLEAKFX_SDK_BIN_DIRS)
list(REVERSE BLEAKFX_SDK_BIN_DIRS)
foreach (dir ${BLEAKFX_SDK_BIN_DIRS})
    list(APPEND BLEAKFX_FXC_HINTS "${dir}/x64")
endforeach ()
find_program(BLEAKFX_FXC NAMES fxc fxc.exe HINTS ${BLEAKFX_FXC_HINTS})

set(BLEAKFX_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/BleakFxShaders.h)
set(BLEAKFX_OBJECTS)

if (BLEAKFX_FXC)
    message(STATUS "BleakFx shaders: ${BLEAKFX_FXC}")
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/shaders)

    foreach (entry ${BLEAKFX_SHADERS})
        string(REPLACE ":" ";" parts ${entry})
        list(GET parts 0 name)
        list(GET parts 1 profile)
        set(object ${CMAKE_CURRENT_BINARY_DIR}/shaders/${name}.cso)
        add_custom_command(
            OUTPUT ${object}
            COMMAND ${BLEAKFX_FXC} /nologo /T ${profile} /E main /O3 /Fo ${object} ${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${name}.hlsl
            DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${name}.hlsl ${BLEAKFX_SHADER_INCLUDES}
            COMMENT "BleakFx: compiling ${name}.hlsl (${profile})"
            VERBATIM
        )
        list(APPEND BLEAKFX_OBJECTS ${object})
    endforeach ()

    # A list cannot pass through one COMMAND argument; EmbedShaders.cmake splits on |.
    string(REPLACE ";" "|" BLEAKFX_OBJECTS_ARG "${BLEAKFX_OBJECTS}")
    add_custom_command(
        OUTPUT ${BLEAKFX_GENERATED}
        COMMAND ${CMAKE_COMMAND} -DOUTPUT=${BLEAKFX_GENERATED} -DINPUTS=${BLEAKFX_OBJECTS_ARG} -P ${CMAKE_CURRENT_SOURCE_DIR}/EmbedShaders.cmake
        DEPENDS ${BLEAKFX_OBJECTS} ${CMAKE_CURRENT_SOURCE_DIR}/EmbedShaders.cmake
        COMMENT "BleakFx: embedding the shaders"
        VERBATIM
    )
elseif (BLEAKFX_REQUIRE_SHADERS)
    message(FATAL_ERROR "fxc (Windows SDK) not found, and BLEAKFX_REQUIRE_SHADERS is on.")
else ()
    message(WARNING "fxc (Windows SDK) not found: BleakFx post-processing is built without shaders and stays off.")
    file(WRITE ${BLEAKFX_GENERATED}
        "// Generated by Engine/Source/BleakFx/CMakeLists.txt: fxc was not found, so there are no shaders.\n"
        "#ifndef BLEAKFX_SHADERS_H\n#define BLEAKFX_SHADERS_H\n\n#define BLEAKFX_HAVE_SHADERS 0\n\n#endif\n")
endif ()

add_library(BleakFx STATIC
    ${PRIVATE_SOURCES}
    ${BLEAKFX_GENERATED}
    ${BLEAKFX_OBJECTS}
)

target_include_directories(BleakFx
    PUBLIC
        ${CMAKE_SOURCE_DIR}
    PRIVATE
        ${CMAKE_CURRENT_BINARY_DIR}
)
