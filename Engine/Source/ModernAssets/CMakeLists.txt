file(GLOB_RECURSE PRIVATE_SOURCES CONFIGURE_DEPENDS "*.cpp")

# BLEAKHOLD(heightblend): WarcraftXL's extended terrain pixel shaders are compiled into the client and served
# in place of the stock ones (see ModernAssets.cpp). The generated source holds one byte array per shader.
set(TERRAIN_SHADER_DIR ${CMAKE_SOURCE_DIR}/ThirdParty/WarcraftXL/shaders/pixel/ps_3_0)
set(TERRAIN_SHADERS Terrain1 Terrain2 Terrain2_pcf Terrain3 Terrain3_pcf)
set(TERRAIN_SHADER_SOURCE ${CMAKE_CURRENT_BINARY_DIR}/ModernAssetsTerrainShaders.cpp)

set(TERRAIN_SHADER_ARRAYS "")
set(TERRAIN_SHADER_TABLE "")

foreach (shader ${TERRAIN_SHADERS})
    set(file ${TERRAIN_SHADER_DIR}/${shader}.bls)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${file})
    file(READ ${file} hex HEX)
    # 32 bytes per line, then each byte as 0xNN.
    string(REGEX REPLACE "([0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f])" "\\1\n" hex "${hex}")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    string(APPEND TERRAIN_SHADER_ARRAYS "static const uint8_t s_${shader}[] = {\n${hex}\n};\n\n")
    string(APPEND TERRAIN_SHADER_TABLE "    { \"${shader}\", s_${shader}, sizeof(s_${shader}) },\n")
endforeach ()

list(LENGTH TERRAIN_SHADERS TERRAIN_SHADER_COUNT)

file(WRITE ${TERRAIN_SHADER_SOURCE}.tmp
"// Generated by Engine/Source/ModernAssets/CMakeLists.txt from ThirdParty/WarcraftXL/shaders. Do not edit.\n\
#include \"Engine/Source/ModernAssets/ModernAssetsShaders.h\"\n\n\
${TERRAIN_SHADER_ARRAYS}\
const ModernAssetsEmbeddedShader g_modernAssetsTerrainShaders[] = {\n${TERRAIN_SHADER_TABLE}};\n\n\
const uint32_t g_modernAssetsTerrainShaderCount = ${TERRAIN_SHADER_COUNT};\n")
# Rewritten only when it changes, so a reconfigure does not rebuild the module.
configure_file(${TERRAIN_SHADER_SOURCE}.tmp ${TERRAIN_SHADER_SOURCE} COPYONLY)

add_library(ModernAssets STATIC
    ${PRIVATE_SOURCES}
    ${TERRAIN_SHADER_SOURCE}
)

# The WarcraftXL sources this calls into use C++20 (std::span).
set_target_properties(ModernAssets PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)

target_include_directories(ModernAssets
    PUBLIC
        ${CMAKE_SOURCE_DIR}
    PRIVATE
        ${CMAKE_SOURCE_DIR}/ThirdParty/WarcraftXL/include
)

target_link_libraries(ModernAssets
    PUBLIC
        BlizzardCore
)
